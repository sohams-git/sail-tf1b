import numpy as np, os

def convert_to_legacy(in_path, out_path, max_ep_len=1000):
    f = np.load(in_path, allow_pickle=True)
    # required inputs
    obs17   = f["observations"].astype(np.float32)     # (N,17)
    acts    = f["actions"].astype(np.float32)          # (N,6)
    rews    = f["rewards"].astype(np.float32).reshape(-1,1)  # (N,1)
    ep_rets = f["episode_returns"]                      # (E,)

    # build episode_starts from episode_lengths if available, else uniform 1000
    if "episode_lengths" in f:
        lens = f["episode_lengths"].astype(int)
    else:
        n = len(obs17)
        E = len(ep_rets)
        lens = np.array([n // E] * E, dtype=int)

    # episode_starts mask and per-step t index
    starts = np.zeros(len(obs17), dtype=bool)
    t_within = np.zeros(len(obs17), dtype=np.int32)
    idx = 0
    for L in lens:
        starts[idx] = True
        t_within[idx:idx+L] = np.arange(L, dtype=np.int32)
        idx += L

    # add time feature to match legacy obs_dim=18 (TimeFeatureWrapper style: 1 - t/1000)
    time_feat = 1.0 - (t_within.astype(np.float32) / float(max_ep_len))
    obs18 = np.concatenate([obs17, time_feat[:,None]], axis=1)

    # write legacy file
    os.makedirs(os.path.dirname(out_path), exist_ok=True)
    np.savez_compressed(
        out_path,
        obs=obs18.astype(np.float32),
        actions=acts.astype(np.float32),
        rewards=rews.astype(np.float32),
        episode_returns=ep_rets.astype(np.float64),
        episode_starts=starts.astype(object)   # matches your original 5600 file dtype
    )

    # quick print
    g = np.load(out_path, allow_pickle=True)
    print(f"[OK] {out_path}")
    print("  keys:", list(g.keys()))
    print("  shapes:", {k:g[k].shape for k in g})
    print("  obs_dim:", g["obs"].shape[1], "act_dim:", g["actions"].shape[1])
    print("  returns min/mean/max:",
          float(g["episode_returns"].min()),
          float(g["episode_returns"].mean()),
          float(g["episode_returns"].max()))
    print()

  Yep — open a **new terminal**. Don’t touch the one that’s training.

Here’s the clean, no-risk way:

1. **In the training terminal (don’t stop it):**

   ```bash
   hostname -f    # note the compute host, e.g. gl1234.arc-ts.umich.edu
   ```

   (This tells you where your job is actually running.)

2. **New SSH session to that same compute node:**

   ```bash
   # if you need your env:
   source ~/.venvs/sail-tf1b/bin/activate   # or module load/conda activate
   # free the port if needed, then start TB
   lsof -ti:6006 | xargs -r kill -9
   tensorboard --logdir ~/sail_logs --host 127.0.0.1 --port 6006 &
   ```

3. **From your laptop, another new terminal to create the tunnel:**

   ```bash
   ssh -N -L 6006:localhost:6006 \
     -J USER@login.greatlakes.arc-ts.umich.edu \
     USER@gl1234.arc-ts.umich.edu
   ```

   Then open: [http://localhost:6006](http://localhost:6006)

Notes:

* If `tensorboard` isn’t found in the new shell, activate the **same env** you used to train, or `pip install tensorboard` there.
* If 6006 is busy, use 6007 (change it in both TB and the `-L` command).
* You can also do this in **VS Code Remote-SSH**: connect to the **compute node**, start TB there, then use the **PORTS** panel → “Forward 6006”.

This way your training keeps running, and you just spin up the viewer and tunnel alongside it.

Pearson correlation (~0.15): measures linear association between episode sums of your learned reward (pred_return) and the true env return. +1 = perfect linear alignment, 0 = no linear relation, −1 = perfect inverse.
→ 0.15 = weak positive linear relation in this setting.
Spearman correlation (~0.11): Pearson computed on the ranks (i.e., it checks if episodes that rank higher by your learned reward also rank higher by true return). It’s insensitive to monotonic rescaling.
→ 0.11 = very weak monotonic agreement (barely above chance) across these 25 random-rollout episodes.
Linear fit T ≈ 0.195·P − 118.7: on average, you can map your learned reward sum P to the env return T with that line. The small slope tells you the two totals vary together only weakly in this distribution.
